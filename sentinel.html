#!/usr/bin/env python3 """ Rename the visible product name Sentinel -> PAYESH across a site / repo. python rename_to_payesh.py . # DRY RUN: shows what would change python rename_to_payesh.py . --write # apply the changes What it changes: the words "Sentinel" / "SENTINEL", links to sentinel.html, and #sentinel anchors. What it deliberately leaves alone: SENTINEL_API_BASE / SENTINEL_API_KEY (secret names), and code identifiers like runSentinel(). Renaming those would break deployments unless you also rename the secrets. Lines marked keep-sentinel are skipped. Anything else still containing "sentinel" is listed at the end for review. """ import re, sys, pathlib EXTS = {'.html', '.py', '.md', '.txt', '.js', '.ts', '.tsx', '.css'} SKIP = {'.git', 'node_modules', '.wrangler', 'dist', '__pycache__'} RULES = [ (re.compile(r'\bSentinel\b'), 'PAYESH'), (re.compile(r'\bSENTINEL\b'), 'PAYESH'), ] def main(): args = [a for a in sys.argv[1:] if not a.startswith('--')] write = '--write' in sys.argv root = pathlib.Path(args[0] if args else '.') me = pathlib.Path(__file__).resolve() changed = left = 0 for p in sorted(root.rglob('*')): if not p.is_file() or p.suffix.lower() not in EXTS or p.resolve() == me: continue if any(part in SKIP for part in p.parts): continue try: text = p.read_text(encoding='utf-8') except UnicodeDecodeError: continue # Lines marked keep-sentinel (or between keep-sentinel:start / :end) are left untouched, # e.g. the "Sentinel is now PAYESH" name-change notice. out, n, in_block = [], 0, False for line in text.splitlines(keepends=True): low = line.lower() if 'keep-sentinel:start' in low: in_block = True if in_block or 'keep-sentinel' in low: out.append(line) else: for pat, rep in RULES: line, k = pat.subn(rep, line) n += k out.append(line) if 'keep-sentinel:end' in low: in_block = False new = ''.join(out) if n: changed += 1 print(f'{"UPDATED" if write else "WOULD UPDATE"} {p} ({n} replacements)') if write: p.write_text(new, encoding='utf-8') for i, line in enumerate(new.splitlines(), 1): if re.search('sentinel', line, re.I): left += 1 print(f' review {p}:{i}: {line.strip()[:110]}') print(f'\n{changed} file(s) {"updated" if write else "would change"}; ' f'{left} line(s) still mention "sentinel" (kept on purpose or need a manual look).') if not write: print('Dry run only. Re-run with --write to apply.') if __name__ == '__main__': main()
